- Use `S3Client.S3ClientConfig` with `endpoint`, `forcePathStyle: true`, and `StaticAWSCredentialIdentityResolver` for Cubbit DS3 compatibility

    import Foundation
    import AWSS3
    import AWSSDKIdentity
    import SmithyIdentity
    import ClientRuntime

    /// Domain-specific errors for S3 operations.
    enum S3Error: Error, LocalizedError {
        case invalidConfig(String)
        case connectionFailed(String)
        case bucketCheckFailed(String)
        case bucketCreationFailed(String)
        case unexpected(Error)

        var errorDescription: String? {
            switch self {
            case .invalidConfig(let msg): return "Invalid configuration: \(msg)"
            case .connectionFailed(let msg): return "Connection failed: \(msg)"
            case .bucketCheckFailed(let msg): return "Bucket check failed: \(msg)"
            case .bucketCreationFailed(let msg): return "Bucket creation failed: \(msg)"
            case .unexpected(let err): return "Unexpected error: \(err.localizedDescription)"
            }
        }
    }

    /// Service layer for S3 operations.
    /// Wraps the aws-sdk-swift S3Client with domain-specific methods.
    struct S3Service {
        private let config: S3Config

        init(config: S3Config) {
            self.config = config
        }

        // MARK: - Client Factory

        /// Creates an S3Client configured for a custom S3-compatible endpoint.
        /// Uses static credentials (access key + secret key) and force-path-style addressing.
        private func makeClient() throws -> S3Client {
            guard config.isValid else {
                throw S3Error.invalidConfig("All credential fields must be non-empty")
            }

            let credentials = AWSCredentialIdentity(
                accessKey: config.accessKey,
                secretKey: config.secretKey
            )
            let credentialResolver = try StaticAWSCredentialIdentityResolver(credentials)

            let clientConfig = try S3Client.S3ClientConfig(
                awsCredentialIdentityResolver: credentialResolver,
                region: config.region,
                endpoint: config.endpointURL,
                forcePathStyle: true   // Required for non-AWS S3-compatible endpoints
            )

            return S3Client(config: clientConfig)
        }

        // MARK: - Connection Test

        /// Tests the S3 connection by calling ListBuckets.
        /// Returns `.success` with available buckets or `.failure` with error details.
        func testConnection() async -> Result<[String], S3Error> {
            do {
                let client = try makeClient()
                let response = try await client.listBuckets(input: ListBucketsInput())
                let bucketNames = response.buckets?.compactMap { $0.name } ?? []
                return .success(bucketNames)
            } catch let error as S3Error {
                return .failure(error)
            } catch {
                return .failure(S3Error.connectionFailed(error.localizedDescription))
            }
        }

        // MARK: - Health Check (lightweight)

        /// A lightweight connectivity check — verifies the endpoint is reachable.
        /// Returns true if the endpoint responds, false otherwise.
        func ping() async -> Bool {
            let result = await testConnection()
            // Even a failure with a specific error means the endpoint is reachable
            // (e.g., auth failure is different from "host unreachable")
            if case .success = result { return true }
            if case .failure(let err) = result {
                // If we got an auth error or similar, the endpoint IS reachable
                // but credentials may be wrong. "Connection failed" wrapping
                // indicates a transport-level issue.
                switch err {
                case .connectionFailed: return false
                case .invalidConfig: return false
                default: return true
                }
            }
            return false
        }
    }
    ```

    // MARK: - Bucket Operations

    /// Checks whether the configured bucket exists on the S3 endpoint.
    /// Uses HeadBucket — returns true if bucket exists, false if 404.
    func bucketExists() async -> Result<Bool, S3Error> {
        do {
            let client = try makeClient()
            let input = HeadBucketInput(bucket: config.bucketName)
            _ = try await client.headBucket(input: input)
            return .success(true)
        } catch let error as S3Error {
            return .failure(error)
        } catch {
            // HeadBucket returns 404 as an error when bucket doesn't exist
            let nsError = error as NSError
            if nsError.domain == "AWSClientRuntime" ||
               nsError.localizedDescription.lowercased().contains("not found") ||
               (nsError.code == 404) {
                return .success(false)
            }
            return .failure(S3Error.bucketCheckFailed(error.localizedDescription))
        }
    }

    /// Creates the configured bucket on the S3 endpoint.
    /// LocationConstraint uses the configured region if applicable.
    func createBucket() async -> Result<Void, S3Error> {
        do {
            let client = try makeClient()
            var input = CreateBucketInput(bucket: config.bucketName)

            // Some S3-compatible endpoints require LocationConstraint for non-us-east-1 regions
            // Use the config's region as the location constraint
            if config.region != "us-east-1" {
                let constraint = BucketLocationConstraint(
                    rawValue: config.region
                )
                input.createBucketConfiguration = CreateBucketConfiguration(
                    locationConstraint: constraint
                )
            }

            _ = try await client.createBucket(input: input)
            return .success(())
        } catch let error as S3Error {
            return .failure(error)
        } catch {
            return .failure(S3Error.bucketCreationFailed(error.localizedDescription))
        }
    }

    /// Ensures the configured bucket exists.
    /// Checks existence first; creates only if missing.
    /// Returns true if bucket already existed, false if newly created.
    func ensureBucketExists() async -> Result<Bool, S3Error> {
        let exists = await bucketExists()
        switch exists {
        case .success(true):
            return .success(true)    // Already exists
        case .success(false):
            // Need to create it
            let created = await createBucket()
            switch created {
            case .success:
                return .success(false)  // Newly created
            case .failure(let error):
                return .failure(error)
            }
        case .failure(let error):
            return .failure(error)
        }
    }
    ```

    This fulfills CONF-02: "App auto-creates the bucket if it doesn't exist on first run."

| T-02-01 | Spoofing | S3Client endpoint | mitigate | Custom endpoint URL is user-provided; app connects to exactly what user configured. User sees endpoint in the UI before testing. No host override. |
| T-02-02 | Information Disclosure | S3Client credentials in error | mitigate | S3Error types do NOT include credential values in error messages. Only user-facing strings like "Connection failed: \(description)". |
