macOS Services integration works via:
1. **Info.plist** with `NSServices` key — registers the service with Finder
2. **NSApplicationDelegate** — implements `application(_:openFiles:)` to receive file paths
3. **Service sends `NSFilenamesPboardType`** — app receives it as `[String]` of file paths

The service sends files as NSPasteboard with `NSFilenamesPboardType` (or the modern `NSPasteboard.ItemDataProvider` approach). The app's delegate receives them via `application(_:openFiles:)`.

Existing app uses SwiftUI App lifecycle (`@main struct IShareApp: App`). To add NSApplicationDelegate conformance, use `NSApplicationDelegateAdaptor`:

```swift
@main
struct IShareApp: App {
    @NSApplicationDelegateAdaptor(AppDelegate.self) var appDelegate
    // ...
}

class AppDelegate: NSObject, NSApplicationDelegate {
    func application(_ sender: NSApplication, openFiles filenames: [String]) {
        // Handle incoming files from Finder Services
    }
}
```

Key design decisions:
- **Info.plist** registers service: name "Share with IShare", accepts files, sends to our app
- **AppDelegate** receives files and publishes to SwiftUI via NotificationCenter or @Published
- **Auto-start share** when file is received from Finder (skip NSOpenPanel, go straight to ShareSheetView)
- **Temp file cleanup** — delete zip files from temp dir after successful upload
- **Single file at a time** — v1 handles one file per share action

    ```swift
    // swift-tools-version: 5.9
    import PackageDescription

    let package = Package(
        name: "IShare",
        platforms: [.macOS(.v14)],
        dependencies: [],
        targets: [
            .executableTarget(
                name: "IShare",
                path: "Sources/IShare",
                resources: [
                    .copy("Info.plist")
                ],
                linkerSettings: [
                    .unsafeFlags(["-Xlinker", "-sectcreate", "-Xlinker", "__TEXT", "-Xlinker", "__info_plist", "-Xlinker", "$(SRCROOT)/Sources/IShare/Info.plist"])
    ```

    IMPORTANT: The Info.plist must be embedded in the final executable for macOS Services to detect it. SPM executable targets don't auto-include Info.plist like Xcode projects do. The linker flags above embed the plist into the `__TEXT,__info_plist` section of the binary, which is what macOS reads to discover registered Services.

    If the linker flags approach doesn't compile cleanly, alternatively place the Info.plist at the package root level and use SPM's `.copy()` resource.

    ```swift
    import SwiftUI
    import UniformTypeIdentifiers

    // MARK: - Notifications

    extension Notification.Name {
        static let shareFileReceived = Notification.Name("com.isare.shareFileReceived")
    }

    // MARK: - App Delegate

    class AppDelegate: NSObject, NSApplicationDelegate {
        func application(_ sender: NSApplication, openFiles filenames: [String]) {
            guard let firstPath = filenames.first else { return }

            let url = URL(fileURLWithPath: firstPath)

            // Ensure the file exists
            guard FileManager.default.fileExists(atPath: url.path) else { return }

            // Post notification to SwiftUI layer
            DispatchQueue.main.async {
                NotificationCenter.default.post(
                    name: .shareFileReceived,
                    object: nil,
                    userInfo: ["fileURL": url]
                )
            }

            // Bring app to foreground
            NSApplication.shared.activate(ignoringOtherApps: true)
        }
    }

    // MARK: - App Entry Point

    @main
    struct IShareApp: App {
        @NSApplicationDelegateAdaptor(AppDelegate.self) var appDelegate
        @StateObject private var configStore = ConfigStore()
        @State private var showShareSheet = false
        @State private var shareFileURL: URL?

        var body: some Scene {
            WindowGroup {
                if configStore.isConfigured {
                    MainContentView(
                        configStore: configStore,
                        onShareFile: { url in
                            shareFileURL = url
                            showShareSheet = true
                        }
                    )
                    .environmentObject(configStore)
                    .sheet(isPresented: $showShareSheet) {
                        if let url = shareFileURL {
                            ShareSheetView(fileURL: url, configStore: configStore)
                        }
                    }
                    .onReceive(NotificationCenter.default.publisher(for: .shareFileReceived)) { notification in
                        if let url = notification.userInfo?["fileURL"] as? URL {
                            shareFileURL = url
                            showShareSheet = true
                        }
                    }
                } else {
                    ConfigView()
                }
            }
            .windowResizability(.contentSize)
            .windowStyle(.hiddenTitleBar)
            .commands {
                CommandGroup(after: .appSettings) {
                    Button("S3 Configuration...") {
                        openSettings()
                    }
                    .keyboardShortcut(",", modifiers: .command)
                }

                CommandGroup(before: .newItem) {
                    Button("Share File...") {
                        openFilePicker()
                    }
                    .keyboardShortcut("n", modifiers: [.command, .shift])
                }
            }
        }

        private func openSettings() {
            let settingsView = SettingsView(configStore: configStore)
            let window = NSWindow(
                contentRect: NSRect(x: 0, y: 0, width: 480, height: 500),
                styleMask: [.titled, .closable, .miniaturizable],
                backing: .buffered,
                defer: false
            )
            window.title = "S3 Configuration"
            window.contentView = NSHostingView(rootView: settingsView)
            window.center()
            window.makeKeyAndOrderFront(nil)
            NSApplication.shared.activate(ignoringOtherApps: true)
        }

        private func openFilePicker() {
            let panel = NSOpenPanel()
            panel.canChooseFiles = true
            panel.canChooseDirectories = true
            panel.allowsMultipleSelection = false
            panel.title = "Select a file to share"

            panel.begin { response in
                if response == .OK, let url = panel.url {
                    shareFileURL = url
                    showShareSheet = true
                }
            }
        }
    }

    // MARK: - Main Content View

    struct MainContentView: View {
        @ObservedObject var configStore: ConfigStore
        let onShareFile: (URL) -> Void

        @State private var showShareSheet = false
        @State private var shareURL: URL?

        var body: some View {
            VStack(spacing: 16) {
                Image(systemName: "checkmark.circle.fill")
                    .font(.system(size: 48))
                    .foregroundStyle(.green)

                Text("Connected to S3")
                    .font(.title)
                    .fontWeight(.semibold)

                Text("Use the Share menu, right-click a file in Finder, or press ⌘⇧N to share via IShare.")
                    .font(.subheadline)
                    .foregroundStyle(.secondary)
                    .multilineTextAlignment(.center)
                    .padding(.horizontal, 40)

                Button {
                    let panel = NSOpenPanel()
                    panel.canChooseFiles = true
                    panel.canChooseDirectories = true
                    panel.allowsMultipleSelection = false
                    panel.title = "Select a file to share"
                    panel.begin { response in
                        if response == .OK, let url = panel.url {
                            shareURL = url
                            showShareSheet = true
                        }
                    }
                } label: {
                    Label("Share a File...", systemImage: "square.and.arrow.up")
                }
                .buttonStyle(.borderedProminent)
                .padding(.top, 8)
            }
            .frame(width: 400, height: 320)
            .sheet(isPresented: $showShareSheet) {
                if let url = shareURL {
                    ShareSheetView(fileURL: url, configStore: configStore)
                }
            }
        }
    }
    ```

    ```swift
        // MARK: - Cleanup

        /// Removes temporary zip files after upload.
        /// Should be called after share completes (success or failure) to clean up.
        /// - Parameter item: The ShareItem that was shared (may have compressed fileURL)
        static func cleanupTempFiles(for item: ShareItem) {
            // If we compressed, the fileURL points to a temp zip file
            if item.compress {
                let tempDir = FileManager.default.temporaryDirectory
                let tempFileURL = item.fileURL
                // Only clean up files in the system temp directory
                if tempFileURL.path.hasPrefix(tempDir.path) {
                    try? FileManager.default.removeItem(at: tempFileURL)
                }
            }
        }
    ```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-03-01 | Tampering | application(_:openFiles:) | mitigate | App only processes the first file path; no shell injection via path since paths come from macOS' own Services system. File existence check gate. |
| T-02-03-02 | Denial of Service | Multiple rapid Services calls | accept | v1 handles one file at a time. Rapid sequential calls may overlap sheets. Acceptable for v1. |
| T-02-03-03 | Information Disclosure | Temp file cleanup | mitigate | Cleanup runs after every share (success or failure). Only removes files in system temp directory. Does not delete original source files. |
| T-02-03-04 | Elevation of Privilege | Finder Services | accept | Services run in app's sandbox context. No elevation possible. |
</threat_model>
