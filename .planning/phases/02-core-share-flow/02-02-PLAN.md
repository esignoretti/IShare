Key design decisions:
- **Services menu in menu bar** for triggering share (not Finder Sync Extension — that requires a separate extension target and full app bundle complexity). macOS Services can be registered via Info.plist `NSServices` key, allowing any Finder right-click → Services → "Share with IShare". This is simpler than a Finder Sync Extension and sufficient for v1.
- **Zip via Process(ditto)** — macOS built-in `/usr/bin/ditto` handles zip compression. No need for ZIP foundation. `ditto -c -k --sequesterRsrc {src} {dest}.zip`
- **Duration picker** — SwiftUI Picker with preset values: 1h (3600), 1d (86400), 7d (604800), 1m (2592000), forever (0)
- **Share sheet** — a SwiftUI sheet presented modally that collects duration, compression preference, and kicks off the share flow
- **ShareItem model** — captures all metadata about a share in progress

For macOS Services integration:
1. Add `CFBundleDocumentTypes` and `NSServices` to Info.plist (handled via Package.swift or a plist file)
2. Register a service called "Share with IShare" that accepts files
3. The app receives the file path via `NSApplicationDelegate` or `onOpenURL`

    ```swift
    import Foundation

    /// Coordinates the end-to-end file share flow: compress → upload → generate URL.
    struct ShareService {
        private let s3Service: S3Service

        init(s3Service: S3Service) {
            self.s3Service = s3Service
        }

        // MARK: - Compression

        /// Compresses a file or directory to a zip archive at the specified output URL.
        /// Uses macOS `/usr/bin/ditto` for zip creation (no external deps).
        /// - Parameters:
        ///   - sourceURL: File or directory to compress
        ///   - outputURL: Destination for the .zip file
        /// - Returns: The output URL on success
        func compress(sourceURL: URL, outputURL: URL) async -> Result<URL, ShareError> {
            let process = Process()
            process.executableURL = URL(fileURLWithPath: "/usr/bin/ditto")
            process.arguments = [
                "-c", "-k", "--sequesterRsrc",
                sourceURL.path,
                outputURL.path
            ]

            // Check that ditto exists
            guard FileManager.default.fileExists(atPath: "/usr/bin/ditto") else {
                return .failure(.compressionFailed("ditto not found at /usr/bin/ditto"))
            }

            return await withCheckedContinuation { continuation in
                process.terminationHandler = { proc in
                    if proc.terminationStatus == 0 {
                        continuation.resume(returning: .success(outputURL))
                    } else {
                        continuation.resume(returning: .failure(
                            .compressionFailed("ditto exited with code \(proc.terminationStatus)")
                        ))
                    }
                }

                do {
                    try process.run()
                } catch {
                    continuation.resume(returning: .failure(.compressionFailed(error.localizedDescription)))
                }
            }
        }

        // MARK: - Share Flow

        /// Runs the complete share flow for a ShareItem.
        /// - Parameter item: The share item (mutated in place via closure-based updates)
        /// - Returns: Updated ShareItem with populated objectKey, presignedURL, and state
        func share(_ item: inout ShareItem, onUpdate: @escaping (ShareItem) -> Void) async {
            var mutableItem = item

            // Step 1: Compress if needed
            if mutableItem.compress {
                mutableItem.state = .compressing
                onUpdate(mutableItem)

                let tempDir = FileManager.default.temporaryDirectory
                let zipName = mutableItem.fileURL.lastPathComponent + ".zip"
                let zipURL = tempDir.appendingPathComponent(zipName)

                let result = await compress(sourceURL: mutableItem.fileURL, outputURL: zipURL)
                switch result {
                case .success(let compressedURL):
                    mutableItem.fileURL = compressedURL
                case .failure(let error):
                    mutableItem.state = .failed(error.localizedDescription)
                    item = mutableItem
                    onUpdate(mutableItem)
                    return
                }
            }

            // Step 2: Upload to S3
            mutableItem.state = .uploading
            onUpdate(mutableItem)

            let uploadResult = await s3Service.uploadFile(
                fileURL: mutableItem.fileURL,
                duration: mutableItem.duration.rawValue,
                progressHandler: { progress in
                    mutableItem.progress = progress
                    onUpdate(mutableItem)
                }
            )

            switch uploadResult {
            case .success(let objectKey):
                mutableItem.objectKey = objectKey
            case .failure(let error):
                mutableItem.state = .failed(error.localizedDescription)
                item = mutableItem
                onUpdate(mutableItem)
                return
            }

            // Step 3: Generate pre-signed URL
            mutableItem.state = .generatingURL
            onUpdate(mutableItem)

            guard let objectKey = mutableItem.objectKey else {
                mutableItem.state = .failed("Missing object key after upload")
                item = mutableItem
                onUpdate(mutableItem)
                return
            }

            // For "forever" duration, use a very long expiry (10 years)
            let expirySeconds = mutableItem.duration.seconds == 0 ? 315_360_000 : mutableItem.duration.seconds

            let presignResult = s3Service.generatePresignedURL(
                objectKey: objectKey,
                durationSeconds: expirySeconds
            )

            switch presignResult {
            case .success(let url):
                mutableItem.presignedURL = url
                mutableItem.state = .complete
            case .failure(let error):
                mutableItem.state = .failed(error.localizedDescription)
            }

            item = mutableItem
            onUpdate(mutableItem)
        }
    }

    /// Errors specific to the share flow.
    enum ShareError: Error, LocalizedError {
        case compressionFailed(String)
        case uploadFailed(String)
        case urlGenerationFailed(String)

        var errorDescription: String? {
            switch self {
            case .compressionFailed(let msg): return "Compression failed: \(msg)"
            case .uploadFailed(let msg): return "Upload failed: \(msg)"
            case .urlGenerationFailed(let msg): return "URL generation failed: \(msg)"
            }
        }
    }
    ```

    - SHAR-03: Single files optionally compress (controlled by ShareItem.compress)
    - Uses macOS `/usr/bin/ditto` — no external dependency
